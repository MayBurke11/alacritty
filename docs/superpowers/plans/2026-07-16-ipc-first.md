# IPC-First Architecture Implementation Plan

> **For agentic workers:** Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Refactor all user-initiated operations through a unified `Action` enum dispatch point in `WindowContext`, making IPC a first-class citizen with JSON-RPC-light protocol.

**Architecture:** Single `Action` enum, single dispatch method `WindowContext::handle_action(Action) -> ActionResult`. Keyboard, menu, and IPC all emit Actions. IPC uses `{id, action, params}` JSON wire format over Unix socket + TCP.

**Tech Stack:** Rust, serde, serde_json, winit, Unix domain sockets, TCP

```rust
use serde::{Deserialize, Serialize};
use serde_json::Value;

#[derive(Serialize, Deserialize, Clone, Debug)]
#[serde(tag = "action", rename_all = "snake_case")]
pub enum Action {
    // Window
    NewWindow {
        #[serde(default, skip_serializing_if = "Option::is_none")]
        command: Option<Vec<String>>,
        #[serde(default, skip_serializing_if = "Option::is_none")]
        cwd: Option<String>,
        #[serde(default, skip_serializing_if = "Option::is_none")]
        config: Option<Value>,
    },
    CloseWindow,

    // Tab
    CreateTab {
        #[serde(default, skip_serializing_if = "Option::is_none")]
        command: Option<Vec<String>>,
        #[serde(default, skip_serializing_if = "Option::is_none")]
        cwd: Option<String>,
        #[serde(default, skip_serializing_if = "Option::is_none")]
        config: Option<Value>,
        #[serde(default)]
        no_switch: bool,
    },
    CloseTab {
        #[serde(default, skip_serializing_if = "Option::is_none")]
        index: Option<usize>,
    },
    SelectTab { index: usize },
    SelectNextTab,
    SelectPreviousTab,
    SelectLastTab,
    ListTabs,
    MoveTab { index: usize, delta: i32 },
    TogglePin { index: usize },
    SetTabTitle { title: String },

    // Pane
    SplitPane { direction: SplitDir },
    ClosePane,
    FocusPane { direction: FocusDir },
    ToggleZoom,
    ResizePane { direction: SplitDir, grow: bool },

    // Session
    SaveSession,
    LoadSession { name: String },
    ListSessions,

    // Config
    GetConfig,
    SetConfig {
        options: HashMap<String, Value>,
        #[serde(default)]
        reset: bool,
    },

    // Shell
    QuickRun {
        command: Vec<String>,
        #[serde(default)]
        no_switch: bool,
    },
    Scroll { lines: i32 },
    Copy,
    Paste { data: String },

    // Menu
    ToggleMenu,
    MenuNavigate { direction: MenuDir },
    MenuSelect,
    MenuBack,
    MenuLetter { ch: char },
    MenuClick { index: usize },

    // Internal
    Bell,

    // Search
    SearchForward,
    SearchBackward,
    SearchNext,
}

#[derive(Serialize, Deserialize, Clone, Copy, Debug)]
#[serde(rename_all = "snake_case")]
pub enum SplitDir {
    Right,
    Down,
}

#[derive(Serialize, Deserialize, Clone, Copy, Debug)]
#[serde(rename_all = "snake_case")]
pub enum FocusDir {
    Left,
    Right,
    Up,
    Down,
}

#[derive(Serialize, Deserialize, Clone, Copy, Debug)]
#[serde(rename_all = "snake_case")]
pub enum MenuDir {
    Left,
    Right,
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct ActionResult {
    pub ok: bool,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub data: Option<Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub error: Option<String>,
}

impl ActionResult {
    pub fn success() -> Self {
        Self { ok: true, data: None, error: None }
    }

    pub fn data(value: Value) -> Self {
        Self { ok: true, data: Some(value), error: None }
    }

    pub fn err(msg: impl Into<String>) -> Self {
        Self { ok: false, data: None, error: Some(msg.into()) }
    }
}
```

```rust
Action::CloseTab { index } => {
    let idx = index.unwrap_or(self.active_tab);
    if idx < self.tabs.len() && !self.tabs[idx].pinned {
        let closing_id = self.tabs[idx].id;
        self.handle_tab_exit(Some(closing_id));
    }
    ActionResult::success()
},
Action::SelectTab { index } => {
    if index < self.tabs.len() {
        self.set_active_tab(index);
    }
    ActionResult::success()
},
Action::SelectNextTab => {
    let next = (self.active_tab + 1) % self.tabs.len().max(1);
    self.set_active_tab(next);
    ActionResult::success()
},
Action::SelectPreviousTab => {
    let prev = if self.active_tab == 0 { self.tabs.len().saturating_sub(1) } else { self.active_tab - 1 };
    self.set_active_tab(prev);
    ActionResult::success()
},
Action::SelectLastTab => {
    if !self.tabs.is_empty() {
        self.set_active_tab(self.tabs.len() - 1);
    }
    ActionResult::success()
},
Action::ListTabs => {
    let tabs: Vec<serde_json::Value> = self.tabs.iter().enumerate().map(|(i, t)| {
        serde_json::json!({
            "index": i + 1,
            "id": t.id.0,
            "title": t.display_title(),
            "active": i == self.active_tab,
            "pinned": t.pinned,
        })
    }).collect();
    ActionResult::data(serde_json::Value::Array(tabs))
},
Action::MoveTab { index, delta } => {
    if index < self.tabs.len() {
        let new = if delta < 0 {
            index.saturating_sub(delta.unsigned_abs())
        } else {
            (index + delta as usize).min(self.tabs.len() - 1)
        };
        self.tabs.swap(index, new);
        if self.active_tab == index { self.active_tab = new; }
        else if self.active_tab == new { self.active_tab = index; }
        self.dirty = true;
    }
    ActionResult::success()
},
Action::TogglePin { index } => {
    if index < self.tabs.len() {
        self.tabs[index].pinned = !self.tabs[index].pinned;
        self.dirty = true;
    }
    ActionResult::success()
},
Action::SetTabTitle { title } => {
    self.active_tab_mut().custom_title = if title.is_empty() { None } else { Some(title) };
    self.dirty = true;
    ActionResult::success()
},
```

```rust
fn translate_config_overrides(config: Option<serde_json::Value>) -> Vec<String> {
    let Some(config) = config else { return Vec::new() };
    let obj = match config {
        serde_json::Value::Object(map) => map,
        _ => return Vec::new(),
    };
    obj.into_iter().map(|(k, v)| {
        match v {
            serde_json::Value::String(s) => format!("{k}={s}"),
            other => format!("{k}={other}"),
        }
    }).collect()
}
```

```rust
use std::net::TcpListener;

pub fn start_tcp_listener(addr: &str, token: Option<String>, proxy: EventLoopProxy<Event>) -> std::io::Result<()> {
    let listener = TcpListener::bind(addr)?;
    std::thread::spawn(move || {
        for stream in listener.incoming().flatten() {
            // Read line, parse IpcRequest, dispatch Action, send IpcResponse
            let mut reader = BufReader::new(&stream);
            let mut line = String::new();
            if reader.read_line(&mut line).is_ok() {
                if let Ok(req) = serde_json::from_str::<IpcRequest>(&line) {
                    let _ = proxy.send_event(Event::new(EventType::Action(req.action), WindowId::from(0)));
                    // Write response
                    let resp = IpcResponse { id: req.id, ok: true, data: None, error: None };
                    let mut json = serde_json::to_string(&resp).unwrap_or_default();
                    json.push('\n');
                    let _ = std::io::Write::write_all(&mut &stream, json.as_bytes());
                }
            }
        }
    });
    Ok(())
}
```
